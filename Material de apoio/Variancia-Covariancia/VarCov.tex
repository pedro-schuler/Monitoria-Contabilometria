\section{Forma da Matriz de Variância-Covariância}
\[
	VarCov =
	\begin{bmatrix}
		Var(\beta_{0})           & Cov(\beta_{0},\beta_{1}) & Cov(\beta_{0},\beta_{2}) & \cdots & Cov(\beta_{0},\beta_{n}) \\
		Cov(\beta_{1},\beta_{0}) & Var(\beta_{1})           & Cov(\beta_{1},\beta_{2}) & \cdots & Cov(\beta_{1},\beta_{n}) \\
		Cov(\beta_{2},\beta_{0}) & Cov(\beta_{2},\beta_{1}) & Var(\beta_{2})           & \cdots & Cov(\beta_{2},\beta_{n}) \\
		\vdots                   & \vdots                   & \vdots                   & \ddots & \vdots                   \\
		Cov(\beta_{n},\beta_{0}) & Cov(\beta_{n},\beta_{1}) & Cov(\beta_{n},\beta_{2}) & \cdots & Var(\beta_{n})
	\end{bmatrix}~
	\begin{array}{c}
		(Var(\beta_{0}) \, e \, Cov(\beta_{0},\beta_{x})) \\
		(Var(\beta_{1}) \, e \, Cov(\beta_{1},\beta_{x})) \\
		(Var(\beta_{2}) \, e \, Cov(\beta_{2},\beta_{x})) \\
		\\
		(Var(\beta_{n}) \, e \, Cov(\beta_{n},\beta_{x})) \\
	\end{array}
\]
